\begin{figure}
    \centering
    \incfig[0.9]{visual-cfar}
    \caption[Visualisation of the CFAR stage.]{Visualisation of the CFAR stage for visual
    validation. The y-axis is time and the x-axis is range. The colour is amplitude
    and no colour (black) indicates that that range bin has a value below the 
    threshold. In this example, there is a target at range "a".}
    \label{fig:cfar-visual}
\end{figure}

\begin{figure}
    \centering
    \includegraphics[width=1\textwidth]{visual-cluster.png} 
    \caption[Visualisation of the clustering stage.]{Visualisation of the clustering 
    stage for visual
    validation. The y-axis is time and the x-axis is range. The colour is amplitude. 
    This time, range is a continuous variable instead of in discrete bins. 
    In this example, there is a \~ -40dB target at around 1 km surrounded 
    by a high level of -47dB clutter.}
    \label{fig:cluster-visual}
\end{figure}

\begin{figure}
    \centering
    \includegraphics[width=0.9\textwidth]{visual-angle.png} 
    \caption[Visualisation of the angle sensing stage.]{Visualisation of the angle 
    sensing stage for visual
    validation. Depicted is a 3D plot of all detections accumulated over time.
    The colour yellow indicates high power detections and purple indicates 
    the lowest power detections. Depicted are two collections of points that
    correspond to targets that were once in those positions. }
    \label{fig:angle-visual}
\end{figure}

\begin{figure}
    \centering
    \includegraphics[width=0.8\textwidth]{visual-track.png} 
    \caption[Visualisation of the tracking stage.]{A 3D plot containing all detections
    tentative tracks, confirmed tracks, and hypotheses accumulated over time with 
    playback controls. 
    Ellipses indicate the uncertainty of the element at its
    centre. For example, white spheres indicate measurement 
    uncertainty, and yellow indicates track update uncertainty. The FOV 
    of each of the five quadruplets is depicted as coloured lines. The image 
    depicts mostly clutter collected over a second, with a cluster of 
    detections at around 1 km from the radar in the red circle. }
    \label{fig:track-visual1}
\end{figure}

\begin{figure}
    \centering
    \includegraphics[width=0.6\textwidth]{visual-track2.png}
    \caption[Tracker Visualisation Zoomed In]{A close up view of 
    the tracking stage identifying a target at around 1 km from the
    radar in high noise conditions. As the ground truth progresses 
    (turquoise line), a tentative track is born (red line, blue dot),
    is updated (yellow dots), and is promoted to a confirmed track.
    Noisy detections (white) are associated with the track (green lines)
    and eventually, the track converges to the target's true state.}
    \label{fig:track-visual2}
\end{figure}
